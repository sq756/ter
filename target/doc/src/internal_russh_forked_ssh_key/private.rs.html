1//! SSH private key support.
2//!
3//! Support for decoding SSH private keys (i.e. digital signature keys)
4//! from the OpenSSH file format:
5//!
6//! <https://cvsweb.openbsd.org/src/usr.bin/ssh/PROTOCOL.key?annotate=HEAD>
7//!
8//! ## Decrypting encrypted private keys
9//!
10//! When the `encryption` feature of this crate is enabled, it's possible to
11//! decrypt keys which have been encrypted under a password:
12//!
13#![cfg_attr(all(feature = "encryption", feature = "std"), doc = " ```")]
14#![cfg_attr(not(all(feature = "encryption", feature = "std")), doc = " ```ignore")]
15//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
16//! use ssh_key::PrivateKey;
17//!
18//! // WARNING: don't actually hardcode private keys in source code!!!
19//! let encoded_key = r#"
20//! -----BEGIN OPENSSH PRIVATE KEY-----
21//! b3BlbnNzaC1rZXktdjEAAAAACmFlczI1Ni1jdHIAAAAGYmNyeXB0AAAAGAAAABBKH96ujW
22//! umB6/WnTNPjTeaAAAAEAAAAAEAAAAzAAAAC3NzaC1lZDI1NTE5AAAAILM+rvN+ot98qgEN
23//! 796jTiQfZfG1KaT0PtFDJ/XFSqtiAAAAoFzvbvyFMhAiwBOXF0mhUUacPUCMZXivG2up2c
24//! hEnAw1b6BLRPyWbY5cC2n9ggD4ivJ1zSts6sBgjyiXQAReyrP35myYvT/OIB/NpwZM/xIJ
25//! N7MHSUzlkX4adBrga3f7GS4uv4ChOoxC4XsE5HsxtGsq1X8jzqLlZTmOcxkcEneYQexrUc
26//! bQP0o+gL5aKK8cQgiIlXeDbRjqhc4+h4EF6lY=
27//! -----END OPENSSH PRIVATE KEY-----
28//! "#;
29//!
30//! let encrypted_key = PrivateKey::from_openssh(encoded_key)?;
31//! assert!(encrypted_key.is_encrypted());
32//!
33//! // WARNING: don't hardcode passwords, and this one's bad anyway
34//! let password = "hunter42";
35//!
36//! let decrypted_key = encrypted_key.decrypt(password)?;
37//! assert!(!decrypted_key.is_encrypted());
38//! # Ok(())
39//! # }
40//! ```
41//!
42//! ## Encrypting plaintext private keys
43//!
44//! When the `encryption` feature of this crate is enabled, it's possible to

49#![cfg_attr(

211    /// Create a new unencrypted private key with the given keypair data and comment.
212    ///
213    /// On `no_std` platforms, use `PrivateKey::from(key_data)` instead.
214    #[cfg(feature = "alloc")]
215    pub fn new(key_data: KeypairData, comment: impl Into<String>) -> Result<Self> {
216        if key_data.is_encrypted() {
217            return Err(Error::Encrypted);

354        #[cfg(not(unix))]

358            .create(true)

360            .truncate(true)

368    /// Attempt to decrypt an encrypted private key using the provided
369    /// password to derive an encryption key.
370    ///
371    /// Returns [`Error::Decrypted`] if the private key is already decrypted.
372    #[cfg(feature = "encryption")]
373    pub fn decrypt(&self, password: impl AsRef<[u8]>) -> Result<Self> {
374        let (key, iv) = self.kdf.derive_key_and_iv(self.cipher, password)?;
375
376        let ciphertext = self.key_data.encrypted().ok_or(Error::Decrypted)?;
377        let mut buffer = Zeroizing::new(ciphertext.to_vec());
378        self.cipher.decrypt(&key, &iv, &mut buffer, self.auth_tag)?;
379
380        Self::decode_privatekey_comment_pair(
381            &mut &**buffer,
382            self.public_key.key_data.clone(),
383            self.cipher.block_size(),
384            self.cipher.block_size() - 1,
385        )
386    }

411        &self,
412        rng: &mut impl CryptoRngCore,
413        cipher: Cipher,
414        password: impl AsRef<[u8]>,
415    ) -> Result<Self> {
416        let checkint = rng.next_u32();

420            Kdf::new(Default::default(), rng)?,
421            checkint,
422            password,

426    /// Encrypt an unencrypted private key using the provided cipher and KDF
427    /// configuration.
428    ///
429    /// Returns [`Error::Encrypted`] if the private key is already encrypted.
430    #[cfg(feature = "encryption")]
431    pub fn encrypt_with(
432        &self,
433        cipher: Cipher,
434        kdf: Kdf,
435        checkint: u32,
436        password: impl AsRef<[u8]>,
437    ) -> Result<Self> {
438        if self.is_encrypted() {
439            return Err(Error::Encrypted);
440        }
441
442        let (key_bytes, iv_bytes) = kdf.derive_key_and_iv(cipher, password)?;
443        let msg_len = self.encoded_privatekey_comment_pair_len(cipher)?;
444        let mut out = Vec::with_capacity(msg_len);
445
446        // Encode and encrypt private key
447        self.encode_privatekey_comment_pair(&mut out, cipher, checkint)?;
448        let auth_tag = cipher.encrypt(&key_bytes, &iv_bytes, out.as_mut_slice())?;
449
450        Ok(Self {
451            cipher,
452            kdf,
453            checkint: None,
454            public_key: self.public_key.key_data.clone().into(),
455            key_data: KeypairData::Encrypted(out),
456            auth_tag,
457        })
458    }

506    /// Generate a random key which uses the given algorithm.
507    ///
508    /// # Returns
509    /// - `Error::AlgorithmUnknown` if the algorithm is unsupported.
510    #[cfg(feature = "rand_core")]
511    #[allow(unreachable_code, unused_variables)]
512    pub fn random(rng: &mut impl CryptoRngCore, algorithm: Algorithm) -> Result<Self> {
513        let checkint = rng.next_u32();
514        let key_data = match algorithm {
515            #[cfg(feature = "dsa")]
516            Algorithm::Dsa => KeypairData::from(DsaKeypair::random(rng)?),
517            #[cfg(any(feature = "p256", feature = "p384", feature = "p521"))]
518            Algorithm::Ecdsa { curve } => KeypairData::from(EcdsaKeypair::random(rng, curve)?),
519            #[cfg(feature = "ed25519")]
520            Algorithm::Ed25519 => KeypairData::from(Ed25519Keypair::random(rng)),
521            #[cfg(feature = "rsa")]
522            Algorithm::Rsa { .. } => {
523                KeypairData::from(RsaKeypair::random(rng, DEFAULT_RSA_KEY_SIZE)?)
524            }
525            _ => return Err(Error::AlgorithmUnknown),
526        };
527        let public_key = public::KeyData::try_from(&key_data)?;
528
529        Ok(Self {
530            cipher: Cipher::None,
531            kdf: Kdf::None,
532            checkint: Some(checkint),
533            public_key: public_key.into(),
534            key_data,
535            auth_tag: None,
536        })
537    }

545    /// Decode [`KeypairData`] along with its associated checkints and comment,
546    /// storing the comment in the provided public key on success.
547    ///
548    /// This method also checks padding for validity and ensures that the
549    /// decoded private key matches the provided public key.
550    ///
551    /// For private key format specification, see OpenSSH [PROTOCOL.key] § 3:
552    ///
553    /// ```text
554    /// uint32  checkint
555    /// uint32  checkint
556    /// byte[]  privatekey1
557    /// string  comment1
558    /// byte[]  privatekey2
559    /// string  comment2
560    /// ...
561    /// string  privatekeyN
562    /// string  commentN
563    /// char    1
564    /// char    2
565    /// char    3
566    /// ...
567    /// char    padlen % 255
568    /// ```
569    ///
570    /// [PROTOCOL.key]: https://cvsweb.openbsd.org/src/usr.bin/ssh/PROTOCOL.key?annotate=HEAD
571    fn decode_privatekey_comment_pair(
572        reader: &mut impl Reader,
573        public_key: public::KeyData,
574        block_size: usize,
575        max_padding_size: usize,
576    ) -> Result<Self> {
577        debug_assert!(block_size <= MAX_BLOCK_SIZE);
578        debug_assert!(max_padding_size <= MAX_BLOCK_SIZE);
579
580        // Ensure input data is padding-aligned
581        if reader.remaining_len().checked_rem(block_size) != Some(0) {
582            return Err(encoding::Error::Length.into());
583        }
584
585        let checkint1 = u32::decode(reader)?;
586        let checkint2 = u32::decode(reader)?;
587
588        if checkint1 != checkint2 {
589            return Err(Error::Crypto);
590        }
591
592        let key_data = KeypairData::decode(reader)?;
593
594        // Ensure public key matches private key
595        if public_key != public::KeyData::try_from(&key_data)? {
596            return Err(Error::PublicKey);
597        }
598
599        let mut public_key = PublicKey::from(public_key);
600        public_key.decode_comment(reader)?;
601
602        let padding_len = reader.remaining_len();
603
604        if padding_len > max_padding_size {
605            return Err(encoding::Error::Length.into());
606        }
607
608        if padding_len != 0 {
609            let mut padding = [0u8; MAX_BLOCK_SIZE];
610            reader.read(&mut padding[..padding_len])?;
611
612            if PADDING_BYTES[..padding_len] != padding[..padding_len] {
613                return Err(Error::FormatEncoding);
614            }
615        }
616
617        if !reader.is_finished() {
618            return Err(Error::TrailingData {
619                remaining: reader.remaining_len(),
620            });
621        }
622
623        Ok(Self {
624            cipher: Cipher::None,
625            kdf: Kdf::None,
626            checkint: Some(checkint1),
627            public_key,
628            key_data,
629            auth_tag: None,
630        })
631    }

633    /// Encode [`KeypairData`] along with its associated checkints, comment,
634    /// and padding.
635    fn encode_privatekey_comment_pair(
636        &self,
637        writer: &mut impl Writer,
638        cipher: Cipher,
639        checkint: u32,
640    ) -> encoding::Result<()> {
641        let unpadded_len = self.unpadded_privatekey_comment_pair_len()?;
642        let padding_len = cipher.padding_len(unpadded_len);
643
644        checkint.encode(writer)?;
645        checkint.encode(writer)?;
646        self.key_data.encode(writer)?;
647        self.comment().encode(writer)?;
648        writer.write(&PADDING_BYTES[..padding_len])?;
649        Ok(())
650    }
651
652    /// Get the length of this private key when encoded with the given comment
653    /// and padded using the padding size for the given cipher.
654    fn encoded_privatekey_comment_pair_len(&self, cipher: Cipher) -> encoding::Result<usize> {
655        let len = self.unpadded_privatekey_comment_pair_len()?;
656        [len, cipher.padding_len(len)].checked_sum()
657    }
658
659    /// Get the length of this private key when encoded with the given comment.
660    ///
661    /// This length is just the checkints, private key data, and comment sans
662    /// any padding.
663    fn unpadded_privatekey_comment_pair_len(&self) -> encoding::Result<usize> {
664        // This method is intended for use with unencrypted keys only
665        debug_assert!(!self.is_encrypted(), "called on encrypted key");
666
667        [
668            8, // 2 x uint32 checkints,
669            self.key_data.encoded_len()?,
670            self.comment().encoded_len()?,
671        ]
672        .checked_sum()
673    }
674}

696impl Decode for PrivateKey {
697    type Error = Error;
698
699    fn decode(reader: &mut impl Reader) -> Result<Self> {
700        let mut auth_magic = [0u8; Self::AUTH_MAGIC.len()];
701        reader.read(&mut auth_magic)?;
702
703        if auth_magic != Self::AUTH_MAGIC {
704            return Err(Error::FormatEncoding);
705        }
706
707        let cipher = Cipher::decode(reader)?;
708        let kdf = Kdf::decode(reader)?;
709        let nkeys = usize::decode(reader)?;
710
711        // TODO(tarcieri): support more than one key?
712        if nkeys != 1 {
713            return Err(encoding::Error::Length.into());
714        }
715
716        let public_key = reader.read_prefixed(public::KeyData::decode)?;
717
718        // Handle encrypted private key
719        #[cfg(not(feature = "alloc"))]
720        if cipher.is_some() {
721            return Err(Error::Encrypted);
722        }
723        #[cfg(feature = "alloc")]
724        if cipher.is_some() {
725            let ciphertext = Vec::decode(reader)?;
726
727            // Ensure ciphertext is padded to the expected length
728            if ciphertext.len().checked_rem(cipher.block_size()) != Some(0) {
729                return Err(Error::Crypto);
730            }
731
732            let auth_tag = if cipher.has_tag() {
733                let mut tag = Tag::default();
734                reader.read(&mut tag)?;
735                Some(tag)
736            } else {
737                None
738            };
739
740            if !reader.is_finished() {
741                return Err(Error::TrailingData {
742                    remaining: reader.remaining_len(),
743                });
744            }
745
746            return Ok(Self {
747                cipher,
748                kdf,
749                checkint: None,
750                public_key: public_key.into(),
751                key_data: KeypairData::Encrypted(ciphertext),
752                auth_tag,
753            });
754        }
755
756        // Processing unencrypted key. No KDF should be set.
757        if kdf.is_some() {
758            return Err(Error::Crypto);
759        }
760
761        reader.read_prefixed(|reader| {
762            // PuTTYgen uses a non-standard block size of 16
763            // and _always_ adds a padding even if data length
764            // is divisible by 16 - for unencrypted keys
765            // in the OpenSSH format.
766            // We're only relaxing the exact length check, but will
767            // still validate that the contents of the padding area.
768            // In all other cases there can be up to (but not including)
769            // `block_size` padding bytes as per `PROTOCOL.key`.
770            let max_padding_size = match cipher {
771                Cipher::None => 16,
772                #[allow(clippy::arithmetic_side_effects)] // block sizes are constants
773                _ => cipher.block_size() - 1,
774            };
775            Self::decode_privatekey_comment_pair(
776                reader,
777                public_key,
778                cipher.block_size(),
779                max_padding_size,
780            )
781        })
782    }
783}

785impl Encode for PrivateKey {
786    fn encoded_len(&self) -> encoding::Result<usize> {
787        let private_key_len = if self.is_encrypted() {
788            self.key_data.encoded_len_prefixed()?
789        } else {
790            [4, self.encoded_privatekey_comment_pair_len(Cipher::None)?].checked_sum()?
791        };
792
793        [
794            Self::AUTH_MAGIC.len(),
795            self.cipher.encoded_len()?,
796            self.kdf.encoded_len()?,
797            4, // number of keys (uint32)
798            self.public_key.key_data().encoded_len_prefixed()?,
799            private_key_len,
800            self.auth_tag.map(|tag| tag.len()).unwrap_or(0),
801        ]
802        .checked_sum()
803    }
804
805    fn encode(&self, writer: &mut impl Writer) -> encoding::Result<()> {
806        writer.write(Self::AUTH_MAGIC)?;
807        self.cipher.encode(writer)?;
808        self.kdf.encode(writer)?;
809
810        // TODO(tarcieri): support for encoding more than one private key
811        1usize.encode(writer)?;
812
813        // Encode public key
814        self.public_key.key_data().encode_prefixed(writer)?;
815
816        // Encode private key
817        if self.is_encrypted() {
818            self.key_data.encode_prefixed(writer)?;
819
820            if let Some(tag) = &self.auth_tag {
821                writer.write(tag)?;
822            }
823        } else {
824            self.encoded_privatekey_comment_pair_len(Cipher::None)?
825                .encode(writer)?;
826
827            let checkint = self.checkint.unwrap_or_else(|| self.key_data.checkint());
828            self.encode_privatekey_comment_pair(writer, Cipher::None, checkint)?;
829        }
830
831        Ok(())
832    }
833}

5// Multiply the 32-bit numbers contained in |lhs| and |rhs|, and store only
6// the upper 32 bits of the resulting product.
7// This implementation was taken from:
8// https://ei1333.github.io/library/math/combinatorics/vectorize-mod-int.hpp.html
9//
10// TODO: Optimize this implementation if performance numbers suggest doing so.
11#[inline(always)]
12fn mulhi_mm256_epi32(lhs: Vec256, rhs: Vec256) -> Vec256 {
13    let prod02 = mm256_mul_epu32(lhs, rhs);
14    let prod13 = mm256_mul_epu32(
15        mm256_shuffle_epi32::<0b11_11_01_01>(lhs),
16        mm256_shuffle_epi32::<0b11_11_01_01>(rhs),
17    );
18
19    mm256_unpackhi_epi64(
20        mm256_unpacklo_epi32(prod02, prod13),
21        mm256_unpackhi_epi32(prod02, prod13),
22    )
23}

25#[inline(always)]
26pub(crate) fn compress_message_coefficient(vector: Vec256) -> Vec256 {
27    let field_modulus_halved = mm256_set1_epi16((FIELD_MODULUS - 1) / 2);
28    let field_modulus_quartered = mm256_set1_epi16((FIELD_MODULUS - 1) / 4);
29
30    let shifted = mm256_sub_epi16(field_modulus_halved, vector);
31    let mask = mm256_srai_epi16::<15>(shifted);
32
33    let shifted_to_positive = mm256_xor_si256(mask, shifted);
34    let shifted_to_positive_in_range =
35        mm256_sub_epi16(shifted_to_positive, field_modulus_quartered);
36
37    mm256_srli_epi16::<15>(shifted_to_positive_in_range)
38}

41#[hax_lib::requires(fstar!(r#"v $COEFFICIENT_BITS >= 0 /\ v $COEFFICIENT_BITS < bits i32_inttype /\
42    range (v ((mk_i32 1) <<! $COEFFICIENT_BITS) - 1) i32_inttype"#))]
43pub(crate) fn compress_ciphertext_coefficient<const COEFFICIENT_BITS: i32>(
44    vector: Vec256,
45) -> Vec256 {
46    let field_modulus_halved = mm256_set1_epi32(((FIELD_MODULUS as i32) - 1) / 2);
47    let compression_factor = mm256_set1_epi32(10_321_340);
48    let coefficient_bits_mask = mm256_set1_epi32((1 << COEFFICIENT_BITS) - 1);
49
50    // ---- Compress the first 8 coefficients ----
51
52    // Take the bottom 128 bits, i.e. the first 8 16-bit coefficients
53    let coefficients_low = mm256_castsi256_si128(vector);
54
55    // If:
56    //
57    // coefficients_low[0:15] = A
58    // coefficients_low[16:31] = B
59    // coefficients_low[32:63] = C
60    // and so on ...
61    //
62    // after this step:
63    //
64    // coefficients_low[0:31] = A
65    // coefficients_low[32:63] = B
66    // and so on ...
67    let coefficients_low = mm256_cvtepi16_epi32(coefficients_low);
68
69    let compressed_low = mm256_slli_epi32::<{ COEFFICIENT_BITS }>(coefficients_low);
70    let compressed_low = mm256_add_epi32(compressed_low, field_modulus_halved);
71
72    let compressed_low = mulhi_mm256_epi32(compressed_low, compression_factor);
73
74    // Due to the mulhi_mm256_epi32 we've already shifted right by 32 bits, we
75    // just need to shift right by 35 - 32 = 3 more.
76    let compressed_low = mm256_srli_epi32::<3>(compressed_low);
77
78    let compressed_low = mm256_and_si256(compressed_low, coefficient_bits_mask);
79
80    // ---- Compress the next 8 coefficients ----
81
82    // Take the upper 128 bits, i.e. the next 8 16-bit coefficients
83    let coefficients_high = mm256_extracti128_si256::<1>(vector);
84    let coefficients_high = mm256_cvtepi16_epi32(coefficients_high);
85
86    let compressed_high = mm256_slli_epi32::<{ COEFFICIENT_BITS }>(coefficients_high);
87    let compressed_high = mm256_add_epi32(compressed_high, field_modulus_halved);
88
89    let compressed_high = mulhi_mm256_epi32(compressed_high, compression_factor);
90    let compressed_high = mm256_srli_epi32::<3>(compressed_high);
91    let compressed_high = mm256_and_si256(compressed_high, coefficient_bits_mask);
92
93    // Combining them, and grouping each set of 64-bits, this function results
94    // in:
95    //
96    // 0: low low low low | 1: high high high high | 2: low low low low | 3: high high high high
97    //
98    // where each |low| and |high| is a 16-bit element
99    let compressed = mm256_packs_epi32(compressed_low, compressed_high);
100
101    // To be in the right order, we need to move the |low|s above in position 2 to
102    // position 1 and the |high|s in position 1 to position 2, and leave the
103    // rest unchanged.
104    mm256_permute4x64_epi64::<0b11_01_10_00>(compressed)
105}

110pub fn decompress_1(a: Vec256) -> Vec256 {
111    let z = mm256_setzero_si256();
112
113    hax_lib::fstar!(
114        r#"
115        assert(Libcrux_intrinsics.Avx2_extract.vec256_as_i16x16 $z == Seq.create 16 (mk_i16 0));
116        assert(forall i. Seq.index (Libcrux_intrinsics.Avx2_extract.vec256_as_i16x16 $z) i == mk_i16 0);
117        assert(forall i. let x = Seq.index (Libcrux_intrinsics.Avx2_extract.vec256_as_i16x16 $a) i in 
118                                      ((0 - v x) == 0 \/ (0 - v x) == -1));
119        assert(forall i. i < 16 ==>
120                        Spec.Utils.is_intb (pow2 15 - 1) 
121                        (0 - v (Seq.index (Libcrux_intrinsics.Avx2_extract.vec256_as_i16x16 $a) i)))
122    "#
123    );
124
125    let s = arithmetic::sub(z, a);
126
127    hax_lib::fstar!(
128        r#"assert(forall i. Seq.index (Libcrux_intrinsics.Avx2_extract.vec256_as_i16x16 $s) i == mk_i16 0 \/ 
129                            Seq.index (Libcrux_intrinsics.Avx2_extract.vec256_as_i16x16 $s) i == mk_i16 (-1))"#
130    );
131
132    arithmetic::bitwise_and_with_constant(s, 1665)
133}

136#[hax_lib::requires(fstar!(r#"v $COEFFICIENT_BITS >= 0 /\ v $COEFFICIENT_BITS < bits i32_inttype"#))]
137pub(crate) fn decompress_ciphertext_coefficient<const COEFFICIENT_BITS: i32>(
138    vector: Vec256,
139) -> Vec256 {
140    let field_modulus = mm256_set1_epi32(FIELD_MODULUS as i32);
141    let two_pow_coefficient_bits = mm256_set1_epi32(1 << COEFFICIENT_BITS);
142
143    // ---- Compress the first 8 coefficients ----
144    let coefficients_low = mm256_castsi256_si128(vector);
145    let coefficients_low = mm256_cvtepi16_epi32(coefficients_low);
146
147    let decompressed_low = mm256_mullo_epi32(coefficients_low, field_modulus);
148    let decompressed_low = mm256_slli_epi32::<1>(decompressed_low);
149    let decompressed_low = mm256_add_epi32(decompressed_low, two_pow_coefficient_bits);
150
151    // We can't shift in one go by (COEFFICIENT_BITS + 1) due to the lack
152    // of support for const generic expressions.
153    let decompressed_low = mm256_srli_epi32::<{ COEFFICIENT_BITS }>(decompressed_low);
154    let decompressed_low = mm256_srli_epi32::<1>(decompressed_low);
155
156    // ---- Compress the next 8 coefficients ----
157    let coefficients_high = mm256_extracti128_si256::<1>(vector);
158    let coefficients_high = mm256_cvtepi16_epi32(coefficients_high);
159
160    let decompressed_high = mm256_mullo_epi32(coefficients_high, field_modulus);
161    let decompressed_high = mm256_slli_epi32::<1>(decompressed_high);
162    let decompressed_high = mm256_add_epi32(decompressed_high, two_pow_coefficient_bits);
163
164    // We can't shift in one go by (COEFFICIENT_BITS + 1) due to the lack
165    // of support for const generic expressions.
166    let decompressed_high = mm256_srli_epi32::<{ COEFFICIENT_BITS }>(decompressed_high);
167    let decompressed_high = mm256_srli_epi32::<1>(decompressed_high);
168
169    // Combining them, and grouping each set of 64-bits, this function results
170    // in:
171    //
172    // 0: low low low low | 1: high high high high | 2: low low low low | 3: high high high high
173    //
174    // where each |low| and |high| is a 16-bit element
175    let compressed = mm256_packs_epi32(decompressed_low, decompressed_high);
176
177    // To be in the right order, we need to move the |low|s above in position 2 to
178    // position 1 and the |high|s in position 1 to position 2, and leave the
179    // rest unchanged.
180    mm256_permute4x64_epi64::<0b11_01_10_00>(compressed)
181}

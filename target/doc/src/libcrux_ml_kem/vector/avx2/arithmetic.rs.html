123pub(crate) fn cond_subtract_3329(vector: Vec256) -> Vec256 {
124    let field_modulus = mm256_set1_epi16(FIELD_MODULUS);
125
126    hax_lib::fstar!(r#"assert (forall i. get_lane $field_modulus i == (mk_i16 3329))"#);
127
128    // Compute v_i - Q and crate a mask from the sign bit of each of these
129    // quantities.
130    let v_minus_field_modulus = mm256_sub_epi16(vector, field_modulus);
131
132    hax_lib::fstar!(
133        "assert (forall i. get_lane $v_minus_field_modulus i == get_lane $vector i -. (mk_i16 3329))"
134    );
135
136    let sign_mask = mm256_srai_epi16::<15>(v_minus_field_modulus);
137
138    hax_lib::fstar!(
139        "assert (forall i. get_lane $sign_mask i == (get_lane $v_minus_field_modulus i >>! (mk_i32 15)))"
140    );
141
142    // If v_i - Q < 0 then add back Q to (v_i - Q).
143    let conditional_add_field_modulus = mm256_and_si256(sign_mask, field_modulus);
144
145    hax_lib::fstar!(
146        r#"assert (forall i. get_lane $conditional_add_field_modulus i == (get_lane $sign_mask i &. (mk_i16 3329)))"#
147    );
148
149    let result = mm256_add_epi16(v_minus_field_modulus, conditional_add_field_modulus);
150
151    hax_lib::fstar!(
152        r#"assert (forall i. get_lane $result i == (get_lane $v_minus_field_modulus i +. get_lane $conditional_add_field_modulus i));
153                     assert (forall i. get_lane $result i == Spec.Utils.cond_sub (get_lane $vector i));
154                     assert (forall i. get_lane $result i == (if (get_lane $vector i) >=. (mk_i16 3329) then get_lane $vector i -! (mk_i16 3329) else get_lane $vector i))"#
155    );

343pub(crate) fn montgomery_reduce_i32s(vec: Vec256) -> Vec256 {
344    let k = mm256_mullo_epi16(
345        vec,

347    );
348    let k_times_modulus = mm256_mulhi_epi16(k, mm256_set1_epi32(FIELD_MODULUS as i32));
349
350    let value_high = mm256_srli_epi32::<16>(vec);
351
352    let result = mm256_sub_epi16(value_high, k_times_modulus);
353
354    let result = mm256_slli_epi32::<16>(result);
355
356    mm256_srai_epi32::<16>(result)
357}

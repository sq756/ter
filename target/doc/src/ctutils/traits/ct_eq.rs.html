8/// Constant-time equality: like `(Partial)Eq` with [`Choice`] instead of [`bool`].
9///
10/// Impl'd for: [`u8`], [`u16`], [`u32`], [`u64`], [`u128`], [`usize`], [`cmp::Ordering`],
11/// [`Choice`], and arrays/slices of any type which also impls [`CtEq`].
12pub trait CtEq<Rhs = Self>
13where
14    Rhs: ?Sized,
15{
16    /// Determine if `self` is equal to `other` in constant-time.
17    fn ct_eq(&self, other: &Rhs) -> Choice;
18
19    /// Determine if `self` is NOT equal to `other` in constant-time.
20    fn ct_ne(&self, other: &Rhs) -> Choice {
21        !self.ct_eq(other)
22    }
23}

25// Impl `CtEq` using the `cmov::CmovEq` trait
26macro_rules! impl_ct_eq_with_cmov_eq {
27    ( $($ty:ty),+ ) => {
28        $(
29            impl CtEq for $ty {
30                #[inline]
31                fn ct_eq(&self, other: &Self) -> Choice {
32                    let mut ret = Choice::FALSE;
33                    self.cmoveq(other, 1, &mut ret.0);
34                    ret
35                }
36            }
37        )+
38    };
39}

77impl<T: CtEq> CtEq for [T] {
78    #[inline]
79    fn ct_eq(&self, other: &[T]) -> Choice {
80        let mut ret = self.len().ct_eq(&other.len());
81        for (a, b) in self.iter().zip(other.iter()) {
82            ret &= a.ct_eq(b);
83        }
84        ret
85    }
86}

88impl<T: CtEq, const N: usize> CtEq for [T; N] {

119    macro_rules! truth_table {
120        ($a:expr, $b:expr, $c:expr) => {
121            assert!($a.ct_eq(&$b).to_bool());
122            assert!(!$a.ct_eq(&$c).to_bool());
123            assert!(!$b.ct_eq(&$c).to_bool());
124
125            assert!(!$a.ct_ne(&$b).to_bool());
126            assert!($a.ct_ne(&$c).to_bool());
127            assert!($b.ct_ne(&$c).to_bool());

129    }

96#[derive(Debug, Clone, Copy)]
97pub(crate) enum ChannelFlushResult {
98    Incomplete {
99        wrote: usize,
100    },
101    Complete {
102        wrote: usize,
103        pending_eof: bool,
104        pending_close: bool,
105    },
106}
107impl ChannelFlushResult {
108    pub(crate) fn wrote(&self) -> usize {
109        match self {
110            ChannelFlushResult::Incomplete { wrote } => *wrote,
111            ChannelFlushResult::Complete { wrote, .. } => *wrote,
112        }
113    }
114    pub(crate) fn complete(wrote: usize, channel: &ChannelParams) -> Self {
115        ChannelFlushResult::Complete {
116            wrote,
117            pending_eof: channel.pending_eof,
118            pending_close: channel.pending_close,
119        }
120    }
121}

124    pub fn newkeys(&mut self, newkeys: NewKeys) {
125        if let Some(ref mut enc) = self.encrypted {
126            enc.exchange = Some(newkeys.exchange);
127            enc.kex = newkeys.kex;
128            enc.key = newkeys.key;
129            enc.client_mac = newkeys.names.client_mac;
130            enc.server_mac = newkeys.names.server_mac;
131            self.remote_to_local = newkeys.cipher.remote_to_local;
132            self.packet_writer
133                .set_cipher(newkeys.cipher.local_to_remote);
134            self.strict_kex = self.strict_kex || newkeys.names.strict_kex();
135
136            // Reset compression state
137            enc.client_compression
138                .init_compress(self.packet_writer.compress());
139            enc.server_compression.init_decompress(&mut enc.decompress);
140        }
141    }

143    pub fn encrypted(&mut self, state: EncryptedState, newkeys: NewKeys) {
144        let strict_kex = newkeys.names.strict_kex();
145        self.encrypted = Some(Encrypted {
146            exchange: Some(newkeys.exchange),
147            kex: newkeys.kex,
148            key: newkeys.key,
149            client_mac: newkeys.names.client_mac,
150            server_mac: newkeys.names.server_mac,
151            session_id: newkeys.session_id,
152            state,
153            channels: HashMap::new(),
154            last_channel_id: Wrapping(1),
155            write: CryptoVec::new(),
156            write_cursor: 0,
157            last_rekey: russh_util::time::Instant::now(),
158            server_compression: newkeys.names.server_compression,
159            client_compression: newkeys.names.client_compression,
160            decompress: crate::compression::Decompress::None,
161            rekey_wanted: false,
162            received_extensions: Vec::new(),
163            extension_info_awaiters: HashMap::new(),
164        });
165        self.remote_to_local = newkeys.cipher.remote_to_local;
166        self.packet_writer
167            .set_cipher(newkeys.cipher.local_to_remote);
168        self.strict_kex = strict_kex;
169
170        // For non-deferred compression (RFC 4253 "zlib"), activate immediately
171        // after initial key exchange. Deferred compression ("zlib@openssh.com")
172        // will be activated later, after authentication succeeds.
173        if let Some(ref mut enc) = self.encrypted {
174            if !enc.client_compression.is_deferred() {
175                enc.client_compression
176                    .init_compress(self.packet_writer.compress());
177            }
178            if !enc.server_compression.is_deferred() {
179                enc.server_compression
180                    .init_decompress(&mut enc.decompress);
181            }
182        }
183    }

185    /// Send a disconnect message.
186    pub fn disconnect(
187        &mut self,
188        reason: Disconnect,
189        description: &str,
190        language_tag: &str,
191    ) -> Result<(), crate::Error> {
192        let disconnect = |buf: &mut CryptoVec| {
193            push_packet!(buf, {
194                msg::DISCONNECT.encode(buf)?;
195                (reason as u32).encode(buf)?;
196                description.encode(buf)?;
197                language_tag.encode(buf)?;
198            });
199            Ok(())
200        };
201        if !self.disconnected {
202            self.disconnected = true;
203            return if let Some(ref mut enc) = self.encrypted {
204                disconnect(&mut enc.write)
205            } else {
206                disconnect(&mut self.packet_writer.buffer().buffer)
207            };
208        }
209        Ok(())
210    }

240impl Encrypted {
241    pub fn byte(&mut self, channel: ChannelId, msg: u8) -> Result<(), crate::Error> {
242        if let Some(channel) = self.channels.get(&channel) {
243            push_packet!(self.write, {
244                self.write.push(msg);
245                channel.recipient_channel.encode(&mut self.write)?;
246            });
247        }
248        Ok(())
249    }

251    pub fn eof(&mut self, channel: ChannelId) -> Result<(), crate::Error> {
252        if let Some(channel) = self.has_pending_data_mut(channel) {
253            channel.pending_eof = true;
254        } else {
255            self.byte(channel, msg::CHANNEL_EOF)?;
256        }
257        Ok(())
258    }
259
260    pub fn close(&mut self, channel: ChannelId) -> Result<(), crate::Error> {
261        if let Some(channel) = self.has_pending_data_mut(channel) {
262            channel.pending_close = true;
263        } else {
264            self.byte(channel, msg::CHANNEL_CLOSE)?;
265            self.channels.remove(&channel);
266        }
267        Ok(())
268    }

270    pub fn sender_window_size(&self, channel: ChannelId) -> usize {
271        if let Some(channel) = self.channels.get(&channel) {
272            channel.sender_window_size as usize
273        } else {
274            0
275        }
276    }
277
278    pub fn adjust_window_size(
279        &mut self,
280        channel: ChannelId,
281        data: &[u8],
282        target: u32,
283    ) -> Result<bool, crate::Error> {
284        if let Some(channel) = self.channels.get_mut(&channel) {
285            trace!(
286                "adjust_window_size, channel = {}, size = {},",
287                channel.sender_channel, target
288            );
289            // Ignore extra data.
290            // https://tools.ietf.org/html/rfc4254#section-5.2
291            if data.len() as u32 <= channel.sender_window_size {
292                channel.sender_window_size -= data.len() as u32;
293            }
294            if channel.sender_window_size < target / 2 {
295                debug!(
296                    "sender_window_size {:?}, target {:?}",
297                    channel.sender_window_size, target
298                );
299                push_packet!(self.write, {
300                    self.write.push(msg::CHANNEL_WINDOW_ADJUST);
301                    channel.recipient_channel.encode(&mut self.write)?;
302                    (target - channel.sender_window_size).encode(&mut self.write)?;
303                });
304                channel.sender_window_size = target;
305                return Ok(true);
306            }
307        }
308        Ok(false)
309    }
310
311    fn flush_channel(
312        write: &mut CryptoVec,
313        channel: &mut ChannelParams,
314    ) -> Result<ChannelFlushResult, crate::Error> {
315        let mut pending_size = 0;
316        while let Some((buf, a, from)) = channel.pending_data.pop_front() {
317            let size = Self::data_noqueue(write, channel, &buf, a, from)?;
318            pending_size += size;
319            if from + size < buf.len() {
320                channel.pending_data.push_front((buf, a, from + size));
321                return Ok(ChannelFlushResult::Incomplete {
322                    wrote: pending_size,
323                });
324            }
325        }
326        Ok(ChannelFlushResult::complete(pending_size, channel))
327    }
328
329    fn handle_flushed_channel(
330        &mut self,
331        channel: ChannelId,
332        flush_result: ChannelFlushResult,
333    ) -> Result<(), crate::Error> {
334        if let ChannelFlushResult::Complete {
335            wrote: _,
336            pending_eof,
337            pending_close,
338        } = flush_result
339        {
340            if pending_eof {
341                self.eof(channel)?;
342            }
343            if pending_close {
344                self.close(channel)?;
345            }
346        }
347        Ok(())
348    }
349
350    pub fn flush_pending(&mut self, channel: ChannelId) -> Result<usize, crate::Error> {
351        let mut pending_size = 0;
352        let mut maybe_flush_result = Option::<ChannelFlushResult>::None;
353
354        if let Some(channel) = self.channels.get_mut(&channel) {
355            let flush_result = Self::flush_channel(&mut self.write, channel)?;
356            pending_size += flush_result.wrote();
357            maybe_flush_result = Some(flush_result);
358        }
359        if let Some(flush_result) = maybe_flush_result {
360            self.handle_flushed_channel(channel, flush_result)?
361        }
362        Ok(pending_size)
363    }
364
365    pub fn flush_all_pending(&mut self) -> Result<(), crate::Error> {
366        for channel in self.channels.values_mut() {
367            Self::flush_channel(&mut self.write, channel)?;
368        }
369        Ok(())
370    }

372    fn has_pending_data_mut(&mut self, channel: ChannelId) -> Option<&mut ChannelParams> {
373        self.channels
374            .get_mut(&channel)
375            .filter(|c| !c.pending_data.is_empty())
376    }

378    pub fn has_pending_data(&self, channel: ChannelId) -> bool {
379        if let Some(channel) = self.channels.get(&channel) {
380            !channel.pending_data.is_empty()
381        } else {
382            false
383        }

386    /// Push the largest amount of `&buf0[from..]` that can fit into
387    /// the window, dividing it into packets if it is too large, and
388    /// return the length that was written.
389    fn data_noqueue(
390        write: &mut CryptoVec,
391        channel: &mut ChannelParams,
392        buf0: &[u8],
393        a: Option<u32>,
394        from: usize,
395    ) -> Result<usize, crate::Error> {
396        if from >= buf0.len() {
397            return Ok(0);
398        }
399        let mut buf = if buf0.len() as u32 > from as u32 + channel.recipient_window_size {
400            #[allow(clippy::indexing_slicing)] // length checked
401            &buf0[from..from + channel.recipient_window_size as usize]
402        } else {
403            #[allow(clippy::indexing_slicing)] // length checked
404            &buf0[from..]
405        };
406        let buf_len = buf.len();
407
408        while !buf.is_empty() {
409            // Compute the length we're allowed to send.
410            let off = std::cmp::min(buf.len(), channel.recipient_maximum_packet_size as usize);
411            match a {
412                None => push_packet!(write, {
413                    write.push(msg::CHANNEL_DATA);
414                    channel.recipient_channel.encode(write)?;
415                    #[allow(clippy::indexing_slicing)] // length checked
416                    buf[..off].encode(write)?;
417                }),
418                Some(ext) => push_packet!(write, {
419                    write.push(msg::CHANNEL_EXTENDED_DATA);
420                    channel.recipient_channel.encode(write)?;
421                    ext.encode(write)?;
422                    #[allow(clippy::indexing_slicing)] // length checked
423                    buf[..off].encode(write)?;
424                }),
425            }
426            trace!(
427                "buffer: {:?} {:?}",
428                write.len(),
429                channel.recipient_window_size
430            );
431            channel.recipient_window_size -= off as u32;
432            #[allow(clippy::indexing_slicing)] // length checked
433            {
434                buf = &buf[off..]
435            }
436        }
437        trace!("buf.len() = {:?}, buf_len = {:?}", buf.len(), buf_len);
438        Ok(buf_len)
439    }

441    pub fn data(
442        &mut self,
443        channel: ChannelId,
444        buf0: CryptoVec,
445        is_rekeying: bool,
446    ) -> Result<(), crate::Error> {
447        if let Some(channel) = self.channels.get_mut(&channel) {
448            assert!(channel.confirmed);
449            if !channel.pending_data.is_empty() && is_rekeying {
450                channel.pending_data.push_back((buf0, None, 0));
451                return Ok(());
452            }
453            let buf_len = Self::data_noqueue(&mut self.write, channel, &buf0, None, 0)?;
454            if buf_len < buf0.len() {
455                channel.pending_data.push_back((buf0, None, buf_len))
456            }
457        } else {
458            debug!("{channel:?} not saved for this session");
459        }
460        Ok(())
461    }

463    pub fn extended_data(
464        &mut self,
465        channel: ChannelId,
466        ext: u32,
467        buf0: CryptoVec,
468        is_rekeying: bool,
469    ) -> Result<(), crate::Error> {
470        if let Some(channel) = self.channels.get_mut(&channel) {
471            assert!(channel.confirmed);
472            if !channel.pending_data.is_empty() && is_rekeying {
473                channel.pending_data.push_back((buf0, Some(ext), 0));
474                return Ok(());
475            }
476            let buf_len = Self::data_noqueue(&mut self.write, channel, &buf0, Some(ext), 0)?;
477            if buf_len < buf0.len() {
478                channel.pending_data.push_back((buf0, Some(ext), buf_len))
479            }
480        }
481        Ok(())
482    }

484    pub fn flush(
485        &mut self,
486        limits: &Limits,
487        writer: &mut PacketWriter,
488    ) -> Result<bool, crate::Error> {
489        // If there are pending packets (and we've not started to rekey), flush them.
490        {
491            while self.write_cursor < self.write.len() {
492                // Read a single packet, encrypt and send it.
493                #[allow(clippy::indexing_slicing)] // length checked
494                let len = BigEndian::read_u32(&self.write[self.write_cursor..]) as usize;
495                #[allow(clippy::indexing_slicing)]
496                let to_write = &self.write[(self.write_cursor + 4)..(self.write_cursor + 4 + len)];
497                trace!("session_write_encrypted, buf = {to_write:?}");
498
499                writer.packet_raw(to_write)?;
500                self.write_cursor += 4 + len
501            }
502        }
503        if self.write_cursor >= self.write.len() {
504            // If all packets have been written, clear.
505            self.write_cursor = 0;
506            self.write.clear();
507        }
508
509        if self.kex.skip_exchange() {
510            return Ok(false);
511        }
512
513        let now = russh_util::time::Instant::now();
514        let dur = now.duration_since(self.last_rekey);
515        Ok(replace(&mut self.rekey_wanted, false)
516            || writer.buffer().bytes >= limits.rekey_write_limit
517            || dur >= limits.rekey_time_limit)
518    }

520    pub fn new_channel_id(&mut self) -> ChannelId {
521        self.last_channel_id += Wrapping(1);
522        while self
523            .channels
524            .contains_key(&ChannelId(self.last_channel_id.0))
525        {
526            self.last_channel_id += Wrapping(1)
527        }
528        ChannelId(self.last_channel_id.0)
529    }
530    pub fn new_channel(&mut self, window_size: u32, maxpacket: u32) -> ChannelId {
531        loop {
532            self.last_channel_id += Wrapping(1);
533            if let std::collections::hash_map::Entry::Vacant(vacant_entry) =
534                self.channels.entry(ChannelId(self.last_channel_id.0))
535            {
536                vacant_entry.insert(ChannelParams {
537                    recipient_channel: 0,
538                    sender_channel: ChannelId(self.last_channel_id.0),
539                    sender_window_size: window_size,
540                    recipient_window_size: 0,
541                    sender_maximum_packet_size: maxpacket,
542                    recipient_maximum_packet_size: 0,
543                    confirmed: false,
544                    wants_reply: false,
545                    pending_data: std::collections::VecDeque::new(),
546                    pending_eof: false,
547                    pending_close: false,
548                });
549                return ChannelId(self.last_channel_id.0);
550            }
551        }
552    }

26#[derive(Debug)]
27#[allow(clippy::large_enum_variant)]
28enum ClientKexState {
29    Created,
30    WaitingForGexReply {
31        names: Names,
32        kex: KexAlgorithm,
33    },
34    WaitingForDhReply {
35        // both KexInit and DH init sent
36        names: Names,
37        kex: KexAlgorithm,
38    },
39    WaitingForNewKeys {
40        server_host_key: PublicKey,
41        newkeys: NewKeys,
42    },
43}
44
45pub(crate) struct ClientKex {
46    exchange: Exchange,
47    cause: KexCause,
48    state: ClientKexState,
49    config: Arc<Config>,
50}

52impl Debug for ClientKex {
53    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
54        let mut s = f.debug_struct("ClientKex");
55        s.field("cause", &self.cause);
56        match self.state {
57            ClientKexState::Created => {
58                s.field("state", &"created");
59            }
60            ClientKexState::WaitingForGexReply { .. } => {
61                s.field("state", &"waiting for GEX response");
62            }
63            ClientKexState::WaitingForDhReply { .. } => {
64                s.field("state", &"waiting for DH response");
65            }
66            ClientKexState::WaitingForNewKeys { .. } => {
67                s.field("state", &"waiting for NEWKEYS");
68            }
69        }
70        s.finish()
71    }
72}

90    pub fn kexinit(&mut self, output: &mut PacketWriter) -> Result<(), Error> {
91        self.exchange.client_kex_init =
92            negotiation::write_kex(&self.config.preferred, output, None)?;

94        Ok(())

97    pub fn step(
98        mut self,
99        input: Option<&mut IncomingSshPacket>,
100        output: &mut PacketWriter,
101    ) -> Result<KexProgress<Self>, Error> {
102        match self.state {
103            ClientKexState::Created => {
104                // At this point we expect to read the KEXINIT from the other side
105
106                let Some(input) = input else {
107                    return Err(Error::KexInit);
108                };
109                if input.buffer.first() != Some(&msg::KEXINIT) {
110                    error!(
111                        "Unexpected kex message at this stage: {:?}",
112                        input.buffer.first()
113                    );
114                    return Err(Error::KexInit);
115                }
116
117                let names = {
118                    // read algorithms from packet.
119                    self.exchange.server_kex_init.extend(&input.buffer);
120                    negotiation::Client::read_kex(
121                        &input.buffer,
122                        &self.config.preferred,
123                        None,
124                        &self.cause,
125                    )?
126                };
127                debug!("negotiated algorithms: {names:?}");
128
129                // seqno has already been incremented after read()
130                if names.strict_kex() && !self.cause.is_rekey() && input.seqn.0 != 1 {
131                    return Err(strict_kex_violation(
132                        msg::KEXINIT,
133                        input.seqn.0 as usize - 1,
134                    ));
135                }
136
137                let mut kex = KEXES.get(&names.kex).ok_or(Error::UnknownAlgo)?.make();
138
139                if kex.skip_exchange() {
140                    // Non-standard no-kex exchange
141                    let newkeys = compute_keys(
142                        CryptoVec::new(),
143                        kex,
144                        names.clone(),
145                        self.exchange.clone(),
146                        self.cause.session_id(),
147                    )?;
148
149                    output.packet(|w| {
150                        msg::NEWKEYS.encode(w)?;
151                        Ok(())
152                    })?;
153
154                    return Ok(KexProgress::Done {
155                        newkeys,
156                        server_host_key: None,
157                    });
158                }
159
160                if kex.is_dh_gex() {
161                    output.packet(|w| {
162                        kex.client_dh_gex_init(&self.config.gex, w)?;
163                        Ok(())
164                    })?;
165
166                    self.state = ClientKexState::WaitingForGexReply { names, kex };
167                } else {
168                    output.packet(|w| {
169                        kex.client_dh(&mut self.exchange.client_ephemeral, w)?;
170                        Ok(())
171                    })?;
172
173                    self.state = ClientKexState::WaitingForDhReply { names, kex };
174                }
175
176                Ok(KexProgress::NeedsReply {
177                    kex: self,
178                    reset_seqn: false,
179                })
180            }
181            ClientKexState::WaitingForGexReply { names, mut kex } => {
182                let Some(input) = input else {
183                    return Err(Error::KexInit);
184                };
185
186                if input.buffer.first() != Some(&msg::KEX_DH_GEX_GROUP) {
187                    error!(
188                        "Unexpected kex message at this stage: {:?}",
189                        input.buffer.first()
190                    );
191                    return Err(Error::KexInit);
192                }
193
194                #[allow(clippy::indexing_slicing)] // length checked
195                let mut r = &input.buffer[1..];
196
197                let prime = Mpint::decode(&mut r)?;
198                let generator = Mpint::decode(&mut r)?;
199                debug!("received gex group: prime={prime}, generator={generator}");
200
201                let group = DhGroup {
202                    prime: prime.as_bytes().to_vec().into(),
203                    generator: generator.as_bytes().to_vec().into(),
204                };
205
206                if group.bit_size() < self.config.gex.min_group_size
207                    || group.bit_size() > self.config.gex.max_group_size
208                {
209                    warn!(
210                        "DH prime size ({} bits) not within requested range",
211                        group.bit_size()
212                    );
213                    return Err(Error::KexInit);
214                }
215
216                let exchange = &mut self.exchange;
217                exchange.gex = Some((self.config.gex.clone(), group.clone()));
218                kex.dh_gex_set_group(group)?;
219                output.packet(|w| {
220                    kex.client_dh(&mut exchange.client_ephemeral, w)?;
221                    Ok(())
222                })?;
223                self.state = ClientKexState::WaitingForDhReply { names, kex };
224
225                Ok(KexProgress::NeedsReply {
226                    kex: self,
227                    reset_seqn: false,
228                })
229            }
230            ClientKexState::WaitingForDhReply { mut names, mut kex } => {
231                // At this point, we've sent ECDH_INTI and
232                // are waiting for the ECDH_REPLY from the server.
233
234                let Some(input) = input else {
235                    return Err(Error::KexInit);
236                };
237
238                if names.ignore_guessed {
239                    // Ignore the next packet if (1) it follows and (2) it's not the correct guess.
240                    debug!("ignoring guessed kex");
241                    names.ignore_guessed = false;
242                    self.state = ClientKexState::WaitingForDhReply { names, kex };
243                    return Ok(KexProgress::NeedsReply {
244                        kex: self,
245                        reset_seqn: false,
246                    });
247                }
248
249                if input.buffer.first()
250                    != Some(match kex.is_dh_gex() {
251                        true => &msg::KEX_DH_GEX_REPLY,
252                        false => &msg::KEX_ECDH_REPLY,
253                    })
254                {
255                    error!(
256                        "Unexpected kex message at this stage: {:?}",
257                        input.buffer.first()
258                    );
259                    return Err(Error::KexInit);
260                }
261
262                #[allow(clippy::indexing_slicing)] // length checked
263                let r = &mut &input.buffer[1..];
264
265                let server_host_key = Bytes::decode(r)?; // server public key.
266                let server_host_key = parse_public_key(&server_host_key)?;
267                debug!(
268                    "received server host key: {:?}",
269                    server_host_key.to_openssh()
270                );
271
272                let server_ephemeral = Bytes::decode(r)?;
273                self.exchange.server_ephemeral.extend(&server_ephemeral);
274                kex.compute_shared_secret(&self.exchange.server_ephemeral)?;
275
276                let mut pubkey_vec = CryptoVec::new();
277                server_host_key.to_bytes()?.encode(&mut pubkey_vec)?;
278
279                let exchange = &self.exchange;
280                let hash = HASH_BUFFER.with({
281                    |buffer| {
282                        let mut buffer = buffer.borrow_mut();
283                        buffer.clear();
284                        kex.compute_exchange_hash(&pubkey_vec, exchange, &mut buffer)
285                    }
286                })?;
287
288                let signature = Bytes::decode(r)?;
289                let signature = Signature::decode(&mut &signature[..])?;
290
291                if let Err(e) = Verifier::verify(&server_host_key, hash.as_ref(), &signature) {
292                    debug!("wrong server sig: {e:?}");
293                    return Err(Error::WrongServerSig);
294                }
295
296                let newkeys = compute_keys(
297                    hash,
298                    kex,
299                    names.clone(),
300                    self.exchange.clone(),
301                    self.cause.session_id(),
302                )?;
303
304                output.packet(|w| {
305                    msg::NEWKEYS.encode(w)?;
306                    Ok(())
307                })?;
308
309                let reset_seqn = newkeys.names.strict_kex() || self.cause.is_strict_rekey();
310
311                self.state = ClientKexState::WaitingForNewKeys {
312                    server_host_key,
313                    newkeys,
314                };
315
316                Ok(KexProgress::NeedsReply {
317                    kex: self,
318                    reset_seqn,
319                })
320            }
321            ClientKexState::WaitingForNewKeys {
322                server_host_key,
323                newkeys,
324            } => {
325                // At this point the exchange is complete
326                // and we're waiting for a KEWKEYS packet
327                let Some(input) = input else {
328                    return Err(Error::KexInit);
329                };
330
331                if input.buffer.first() != Some(&msg::NEWKEYS) {
332                    error!(
333                        "Unexpected kex message at this stage: {:?}",
334                        input.buffer.first()
335                    );
336                    return Err(Error::Kex);
337                }
338
339                Ok(KexProgress::Done {
340                    newkeys,
341                    server_host_key: Some(server_host_key),
342                })
343            }
344        }
345    }
346}

348fn compute_keys(
349    hash: CryptoVec,
350    kex: KexAlgorithm,
351    names: Names,
352    exchange: Exchange,
353    session_id: Option<&CryptoVec>,
354) -> Result<NewKeys, Error> {
355    let session_id = if let Some(session_id) = session_id {
356        session_id
357    } else {
358        &hash
359    };
360    // Now computing keys.
361    let c = kex.compute_keys(
362        session_id,
363        &hash,
364        names.cipher,
365        names.server_mac,
366        names.client_mac,
367        false,
368    )?;
369    Ok(NewKeys {
370        exchange,
371        names,
372        kex,
373        key: 0,
374        cipher: c,
375        session_id: session_id.clone(),
376    })

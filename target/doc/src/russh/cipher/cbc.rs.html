35impl<C: BlockEncrypt + BlockCipher + BlockDecrypt> BlockStreamCipher for CbcWrapper<C> {
36    fn encrypt_data(&mut self, data: &mut [u8]) {
37        for chunk in data.chunks_exact_mut(C::block_size()) {
38            let mut block = generic_array_from_slice(chunk);
39            self.encryptor.encrypt_block_mut(&mut block);
40            chunk.copy_from_slice(&block);
41        }
42    }
43
44    fn decrypt_data(&mut self, data: &mut [u8]) {
45        for chunk in data.chunks_exact_mut(C::block_size()) {
46            let mut block = generic_array_from_slice(chunk);
47            self.decryptor.decrypt_block_mut(&mut block);
48            chunk.copy_from_slice(&block);
49        }
50    }
51}
